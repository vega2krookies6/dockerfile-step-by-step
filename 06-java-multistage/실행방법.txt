# (재실행 시 기존 컨테이너 정리)
# - 같은 이름(my-java)의 컨테이너가 있으면 run 단계에서 이름 충돌 오류 발생
# docker rm -f my-java

# 이미지 빌드
# - 현재 폴더(java-app)에 Main.java와 Dockerfile이 있어야 함
docker build -t java-app .

# 컨테이너 실행
# - -d: 백그라운드 실행, -p 8080:8080: 내 PC 8080 포트를 컨테이너 8080 포트에 연결
# - --name: 이후 stop/rm/logs 등에서 이 이름으로 관리
docker run -d -p 8080:8080 --name my-java java-app

# 테스트 방법 (브라우저 주소창에 입력)
# - 접속이 안 되면 docker ps로 실행 상태, docker logs my-java로 로그 확인
http://localhost:8080/          # "Hello from Java 17 in Docker!" 출력
http://localhost:8080/health    # "OK" 출력
