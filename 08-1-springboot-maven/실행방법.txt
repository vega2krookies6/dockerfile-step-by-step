# (재실행 시 기존 컨테이너 정리)
# - 같은 이름(my-spring)의 컨테이너가 있으면 run 단계에서 이름 충돌 오류 발생
# docker rm -f my-spring

# 이미지 빌드
# - 처음에는 Maven이 라이브러리를 내려받느라 몇 분 걸릴 수 있음
# - 두 번째부터는 pom.xml이 바뀌지 않는 한 캐시를 사용해 빨라짐
docker build -t spring-boot-app .

# 컨테이너 실행
# - -d: 백그라운드 실행
# - -p 8091:8080: 내 PC의 8091 포트를 컨테이너의 8080 포트(Tomcat)에 연결
#   (6단계 Java 예제의 8080 포트와 겹치지 않도록 8091 사용)
# - --name: 이후 stop/rm/logs 등에서 이 이름으로 관리
docker run -d -p 8091:8080 --name my-spring spring-boot-app

# 테스트 방법 (브라우저 주소창에 입력)
# - 접속이 안 되면 docker logs my-spring으로 시작 로그 확인 (Started DemoApplication 메시지)
http://localhost:8091/                   # 기본 경로, 텍스트 응답
http://localhost:8091/hello/World        # 경로 변수 테스트, JSON 응답
http://localhost:8091/actuator/health    # 상태 확인, {"status":"UP"}이면 정상

# 건강 상태 확인
# - starting → healthy로 바뀌는지 확인 (시작 후 30초 정도 소요)
docker inspect --format="Health Status: {{.State.Health.Status}}" my-spring
