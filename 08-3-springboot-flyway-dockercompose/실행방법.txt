# ==========================================================
# 08-3-springboot-flyway-dockercompose 실행 방법
# - MariaDB + Spring Boot를 Docker Compose로 한 번에 실행
# - 테이블은 Flyway가 만들고, JPA는 ddl-auto=validate로 검사만 함
# - 모든 명령은 08-3 폴더 안에서 실행
# - Compose는 경로를 파일에서 읽으므로 Git Bash, PowerShell 명령이 모두 같음
#   (08-2의 ${PWD}, $(pwd -W), MSYS_NO_PATHCONV=1 이 필요 없음)
# ==========================================================

# 0. 사전 준비 (처음 한 번)
# - .env.example을 .env로 복사하고 비밀번호 입력
# - 08-2와 포트가 겹치지 않으므로(앱 8093, DB 3307) 08-2를 끄지 않아도 됨
cp .env.example .env                    # Git Bash
Copy-Item .env.example .env             # PowerShell

# 1. 설정 확인 (선택)
# - compose.yaml에 .env 값이 치환된 최종 설정을 보여줌
# - 오타나 들여쓰기 오류가 있으면 여기서 바로 알려줌
# - app 서비스의 environment에 MARIADB_ROOT_PASSWORD가 없는 것도 확인
docker compose config

# 2. 빌드 + 실행
# - up: compose.yaml의 서비스를 모두 만들고 시작 (네트워크, 볼륨도 자동 생성)
# - -d: 백그라운드 실행
# - --build: 앱 이미지를 (다시) 빌드한 뒤 실행 (코드를 고쳤을 때도 이 명령)
# - 순서: MariaDB 시작 → healthy가 될 때까지 대기 → 앱 시작 (depends_on)
# - 처음에는 MariaDB 이미지 다운로드 + Maven 빌드로 몇 분 걸림
docker compose up -d --build

# 3. 상태 확인
# - mariadb가 (healthy)가 된 뒤 app이 시작되는지 확인
# - 컨테이너 이름은 boot-flyway-mariadb-1, boot-flyway-app-1 형태
docker compose ps

# 4. 로그 확인
# - 서비스 이름(app, mariadb)으로 지정 (컨테이너 이름이 아님)
# - -f: 실시간으로 계속 보기 (Ctrl + C로 빠져나와도 컨테이너는 계속 실행)
docker compose logs -f app
#   Flyway 로그 (처음 실행 시):
#     Migrating schema `boot_db` to version "1 - create memo table"
#     Migrating schema `boot_db` to version "2 - insert welcome memo"
#     Successfully applied 2 migrations to schema `boot_db`, now at version v2
#   이후 재시작 시:
#     Schema `boot_db` is up to date. No migration necessary.

# MariaDB 초기화 SQL이 실행됐는지 확인
docker compose logs mariadb | grep -E "01-revoke|ERROR"                       # Git Bash
docker compose logs mariadb | Select-String "01-revoke|ERROR"                  # PowerShell

# 5. 테스트 (브라우저)
http://localhost:8093/
http://localhost:8093/memos                # V2가 넣은 첫 메모 1개가 보여야 함
http://localhost:8093/actuator/health      # components.db.status가 UP이면 정상

# 6. 테스트 (메모 저장, 조회, 삭제)
# (Git Bash)
curl -X POST http://localhost:8093/memos -H "Content-Type: application/json" -d '{"content":"compose memo"}'
curl http://localhost:8093/memos
curl -X DELETE http://localhost:8093/memos/2

# (PowerShell)
Invoke-RestMethod -Method Post -Uri http://localhost:8093/memos -ContentType "application/json" -Body '{"content":"compose memo"}'
Invoke-RestMethod http://localhost:8093/memos
Invoke-RestMethod -Method Delete -Uri http://localhost:8093/memos/2

# 7. Flyway 적용 기록 확인
# - docker compose exec 서비스이름 명령: 실행 중인 컨테이너 안에서 명령 실행
# - 비밀번호는 .env의 MARIADB_PASSWORD
docker compose exec mariadb mariadb -uboot -p boot_db -e "SELECT installed_rank, version, description, success FROM flyway_schema_history;"

# 8. 테이블 구조와 권한 확인
docker compose exec mariadb mariadb -uboot -p boot_db -e "SHOW CREATE TABLE memo\G"
docker compose exec mariadb mariadb -uboot -p -e "SHOW GRANTS;"                       # DROP이 없어야 정상

# 9. 코드를 고쳤을 때 (앱만 다시 빌드)
# - 서비스 이름을 지정하면 그 서비스만 다시 빌드하고 교체 (MariaDB는 그대로)
docker compose up -d --build app

# 10. 중지와 삭제
# - stop: 컨테이너만 멈춤 (docker compose start 로 다시 켬)
# - down: 컨테이너와 네트워크 삭제, 볼륨(데이터)은 남음
# - down -v: 볼륨까지 삭제 → 다음 up에서 DB 자동 초기화 + Flyway V1, V2를 처음부터 다시 적용
docker compose stop
docker compose down
docker compose down -v

# ==========================================================
# (실습) 새 마이그레이션 추가해 보기: memo에 updated_at 컬럼 추가
# - 08-3을 한 번 실행해서 V1, V2가 적용된 상태에서 시작
# ==========================================================

# 1) 마이그레이션 파일 생성
#    위치: src/main/resources/db/migration/V3__add_updated_at.sql  (V3 뒤는 밑줄 2개)
#    내용:
#      -- V3: 수정 시각 컬럼 추가

#      -- 이미 있는 메모(V2가 넣은 첫 메모 등)는 작성 시각으로 채움 (데이터 마이그레이션)
#      UPDATE memo SET updated_at = created_at WHERE updated_at IS NULL;

# 2) Memo.java 수정
#    (1) 필드 추가 (createdAt 필드 아래)
#      @Column(name = "updated_at")
#      private LocalDateTime updatedAt;
#
#    (2) getter 추가
#      - JPA(validate)에는 필요 없지만, 없으면 /memos 응답 JSON에 updatedAt이 나오지 않음
#      public LocalDateTime getUpdatedAt() {
#          return updatedAt;
#      }
#
#    (3) (선택) 새 메모를 저장할 때 값 채우기: 생성자에 한 줄 추가
#      - 안 하면 새로 저장하는 메모는 updatedAt이 null

#          this.content = content;
#          this.createdAt = LocalDateTime.now();
#          this.updatedAt = this.createdAt;      // ← 추가
#      }
#
#    (4) (선택) 수정될 때 자동으로 시각 갱신
#      - 지금은 수정 API가 없어서 호출되지 않음, 나중에 PUT /memos/{id}를 만들면 동작
#      - import jakarta.persistence.PreUpdate; 추가 필요
#      @PreUpdate
#      void onUpdate() {
#          this.updatedAt = LocalDateTime.now();
#      }

# 3) 앱만 다시 빌드, 교체 (MariaDB와 데이터는 그대로)
docker compose up -d --build app

# 4) 로그에서 V3 적용 확인
docker compose logs -f app
#   Migrating schema `boot_db` to version "3 - add updated at"
#   Successfully applied 1 migration to schema `boot_db`, now at version v3
#   (V1, V2는 이미 적용됐으므로 다시 실행되지 않음)

# 5) 결과 확인
# - 기존 메모: updatedAt = createdAt (V3의 UPDATE가 채움)
# - 새 메모: (3)을 했다면 값이 있고, 안 했다면 null
curl http://localhost:8093/memos                                                          # Git Bash
curl -X POST http://localhost:8093/memos -H "Content-Type: application/json" -d '{"content":"after v3"}'
Invoke-RestMethod http://localhost:8093/memos                                             # PowerShell

# 적용 기록에 3번이 추가됐는지, 테이블에 컬럼이 생겼는지 확인
docker compose exec mariadb mariadb -uboot -p boot_db -e "SELECT installed_rank, version, description, success FROM flyway_schema_history;"
docker compose exec mariadb mariadb -uboot -p boot_db -e "SHOW CREATE TABLE memo\G"

# (일부러 실패해 보기) 실패 확인 후에는 반드시 원래대로 되돌릴 것
# ----------------------------------------------------------
#
# [A] Entity만 바꾸고 마이그레이션을 빼면
#   - V3 파일을 만들기 전에 2)만 하고 빌드
#   - 결과: 앱이 시작되지 않음 (docker compose ps에서 app이 계속 재시작)
#       Schema-validation: missing column [updated_at] in table [memo]
#   - 의미: ddl-auto=validate가 "코드와 DB가 다르다"를 시작 단계에서 막아 줌
#   - 복구: V3 파일을 추가하고 다시 docker compose up -d --build app
#
# [B] 이미 적용된 파일을 수정하면
#   - V1__create_memo_table.sql에 주석 한 줄만 추가해도 해당
#   - 결과: 앱이 시작되지 않음
#       Validate failed: Migrations have failed validation
#       Migration checksum mismatch for migration version 1
#   - 의미: 적용된 마이그레이션은 기록(체크섬)과 달라지면 안 됨
#   - 복구: V1 파일을 원래 내용으로 되돌리고 다시 빌드
#
# [C] 마이그레이션 SQL에 오류가 있으면
#   - 예: V3에서 컬럼 이름 오타 (UPDATE memo SET updatd_at = ...)
#   - 결과: 앱이 시작되지 않음, flyway_schema_history에 version 3이 success = 0 으로 기록
#   - 주의: MariaDB는 ALTER TABLE을 되돌리지 못하므로(트랜잭션 불가) 앞 문장은 이미 적용된 상태일 수 있음
#   - 복구 (실습용): 데이터까지 지우고 처음부터 다시 적용
#       docker compose down -v
#       docker compose up -d --build
#     (실제 운영 DB에서는 적용된 부분을 직접 정리한 뒤 실패 기록을 지우는 flyway repair를 사용)
