# (재실행 시 기존 컨테이너 정리)
# - 같은 이름(my-react)의 컨테이너가 있으면 run 단계에서 이름 충돌 오류 발생
# docker rm -f my-react

# 이미지 빌드
# - 1단계에서 npm install과 vite build가 실행되어 처음에는 시간이 조금 걸림
# - 현재 폴더에 Dockerfile, package.json, index.html, src/가 있어야 함
docker build -t react-vite-app .

# 컨테이너 실행
# - -d: 백그라운드 실행
# - -p 8090:80: 내 PC의 8090 포트를 컨테이너의 80 포트(Nginx)에 연결
#   (앞 단계 예제들과 포트가 겹치지 않도록 8090 사용)
# - --name: 이후 stop/rm/logs 등에서 이 이름으로 관리
docker run -d -p 8090:80 --name my-react react-vite-app

# 웹사이트 확인: http://localhost:8090
# - 버튼을 누를 때마다 숫자가 올라가면 정상
# - 접속이 안 되면 docker ps로 실행 상태, docker logs my-react로 로그 확인

# (참고) Docker 없이 로컬에서 개발 모드로 실행
# - PC에 Node 22가 설치되어 있어야 함
# - 코드를 저장하면 브라우저에 바로 반영됨 (http://localhost:5173)
npm install
npm run dev
