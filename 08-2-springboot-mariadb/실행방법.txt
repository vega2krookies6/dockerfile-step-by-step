# ==========================================================
# 08-2-springboot-mariadb 실행 방법
# - 컨테이너 2개(MariaDB, Spring Boot)를 같은 Docker 네트워크로 연결
# - 모든 명령은 08-2 폴더 안에서 실행
# - 2단계(MariaDB 실행)는 Git Bash용과 PowerShell용이 다르고, 나머지는 둘 다 같음
# ==========================================================

# 0. 사전 준비 (처음 한 번)
# - .env.example을 .env로 복사하고 비밀번호 입력
# - 이전에 만든 mariadb 컨테이너가 있으면 이름이 겹치므로 삭제
#   (데이터는 이전 볼륨 mariadb-data에 그대로 남아 있음)
docker rm -f mariadb

# 1. 네트워크 만들기 (처음 한 번)
# - 같은 네트워크에 있는 컨테이너끼리는 "컨테이너 이름"으로 서로 접속 가능
# - 앱은 DB_HOST=mariadb 로 MariaDB 컨테이너를 찾음
docker network create boot-net

# 2. MariaDB 실행
#
# [옵션 설명]
#
# MSYS_NO_PATHCONV=1
#   Git Bash 전용. 이 명령 한 번에만 "경로 자동 변환"을 끔
#   - Git Bash는 /로 시작하는 인자를 Windows 경로로 바꿔 버림
#     (/docker-entrypoint-initdb.d → C:/Program Files/Git/docker-entrypoint-initdb.d)
#   - 컨테이너 안의 리눅스 경로가 바뀌지 않도록 막는 역할
#   - PowerShell은 이런 변환을 하지 않으므로 필요 없음
#
# docker run
#   이미지로 새 컨테이너를 만들고 실행 (이미지가 없으면 자동으로 내려받음)
#
# -d
#   백그라운드 실행 (터미널을 점유하지 않음, 출력은 docker logs로 확인)
#
# --name mariadb
#   컨테이너 이름 지정
#   - 앱 컨테이너가 이 이름으로 DB에 접속함 (Dockerfile의 DB_HOST=mariadb)
#   - 바꾸면 앱 실행 시 -e DB_HOST=새이름 도 함께 지정해야 함
#
# --network boot-net
#   1단계에서 만든 네트워크에 연결
#   - 같은 네트워크의 컨테이너끼리는 "컨테이너 이름"을 주소로 서로 접속 가능
#
# -p 127.0.0.1:3306:3306
#   내 PC의 3306 포트 → 컨테이너의 3306 포트 연결
#   - 127.0.0.1: 내 PC에서만 접속 허용 (같은 네트워크의 다른 PC는 차단)
#   - DBeaver 같은 내 PC 프로그램용이며, 앱 컨테이너는 이 옵션 없이도 접속됨
#   - PC에 MySQL/MariaDB가 이미 설치되어 있으면 127.0.0.1:3307:3306 처럼 앞 번호만 변경
#
# --restart unless-stopped
#   컨테이너가 오류로 종료되거나 Docker Desktop을 다시 켜면 자동으로 다시 시작
#   - docker stop으로 직접 멈춘 경우에는 다시 시작하지 않음
#   - always는 직접 멈춰도 Docker를 다시 켜면 또 시작됨
#
# --env-file .env
#   .env 파일의 값을 환경 변수로 전달 (비밀번호를 명령에 직접 쓰지 않기 위해)
#   - MARIADB_ROOT_PASSWORD : root 비밀번호
#   - MARIADB_DATABASE      : 자동으로 만들 DB (boot_db)
#   - MARIADB_USER          : 자동으로 만들 계정 (boot, boot_db에 ALL PRIVILEGES)
#   - MARIADB_PASSWORD      : 그 계정의 비밀번호
#   - 볼륨이 비어 있을 때(처음 시작할 때)만 적용됨
#
# -v boot-mariadb-data:/var/lib/mysql
#   이름 있는 볼륨(boot-mariadb-data)을 MariaDB 데이터 폴더에 연결
#   - 컨테이너를 지워도 DB, 테이블, 데이터가 볼륨에 남음
#   - 볼륨이 없으면 처음 실행할 때 자동으로 만들어짐
#
# -v "$(pwd -W)/db/init:/docker-entrypoint-initdb.d"      (Git Bash)
# -v "${PWD}\db\init:/docker-entrypoint-initdb.d"          (PowerShell)
#   내 PC의 db/init 폴더를 컨테이너의 초기화 폴더에 연결 (바인드 마운트)
#   - MariaDB는 처음 시작할 때 이 폴더의 .sql 파일을 실행 → 01-revoke-drop.sql (DROP 권한 회수)
#   - 상대 경로(./db/init)는 Windows에서 볼륨 이름으로 잘못 해석될 수 있어 절대 경로 사용
#   - Git Bash의 $(pwd)는 /c/... 형태라 Docker가 인식 못 함 → -W를 붙여 C:/... 형태로
#   - 큰따옴표: 폴더 경로에 공백이나 한글이 있어도 하나의 값으로 전달
#
# --health-cmd "healthcheck.sh --connect --innodb_initialized"
#   Dockerfile 없이 실행 시점에 HEALTHCHECK 지정
#   - healthcheck.sh: MariaDB 공식 이미지에 들어 있는 검사 스크립트
#   - --connect: DB에 실제로 접속이 되는지 확인
#   - --innodb_initialized: 저장 엔진(InnoDB) 초기화가 끝났는지 확인
#   - docker ps의 STATUS가 (healthy)가 되면 앱을 실행해도 됨
#
# --health-interval 10s
#   10초마다 검사 (기본값 30s보다 짧게 해서 준비 완료를 빨리 알 수 있게)
#
# --health-start-period 30s
#   시작 후 30초 동안의 검사 실패는 unhealthy로 세지 않음 (첫 초기화에 시간이 걸리므로)
#
# mariadb:10.11.18
#   사용할 이미지와 버전 (10.11은 장기 지원 LTS 버전)
#   - 이 뒤에 오는 값은 docker 옵션이 아니라 MariaDB 서버에 전달되는 옵션
#
# --character-set-server=utf8mb4
#   서버 기본 문자셋을 utf8mb4로 (한글, 이모지까지 저장 가능)
#
# --collation-server=utf8mb4_unicode_ci
#   문자열 비교, 정렬 규칙 (ci: 대소문자 구분 없음)
#
# [줄 바꿈 주의] (Git Bash)
#   - 줄 끝의 \ 는 "명령이 다음 줄로 이어진다"는 뜻
#   - \ 뒤에 공백이 있으면 줄이 끊겨 오류가 나므로, \ 가 반드시 그 줄의 마지막 글자여야 함
#   - 줄 중간에 주석(#)을 넣을 수 없으므로 설명은 위에 따로 적음

# (Git Bash)
MSYS_NO_PATHCONV=1 docker run -d \
  --name mariadb \
  --network boot-net \
  -p 127.0.0.1:3306:3306 \
  --restart unless-stopped \
  --env-file .env \
  -v boot-mariadb-data:/var/lib/mysql \
  -v "$(pwd -W)/db/init:/docker-entrypoint-initdb.d" \
  --health-cmd "healthcheck.sh --connect --innodb_initialized" \
  --health-interval 10s \
  --health-start-period 30s \
  mariadb:10.11.18 \
  --character-set-server=utf8mb4 \
  --collation-server=utf8mb4_unicode_ci

# (PowerShell Only)
docker run -d --name mariadb --network boot-net -p 127.0.0.1:3306:3306 --restart unless-stopped --env-file .env -v boot-mariadb-data:/var/lib/mysql -v "${PWD}\db\init:/docker-entrypoint-initdb.d" --health-cmd "healthcheck.sh --connect --innodb_initialized" --health-interval 10s --health-start-period 30s mariadb:10.11.18 --character-set-server=utf8mb4 --collation-server=utf8mb4_unicode_ci

# MariaDB Volume 확인
docker volume ls

# (초기화 SQL이 실행됐는지 확인)
# - "running /docker-entrypoint-initdb.d/01-revoke-drop.sql"이 보이고 ERROR가 없으면 정상
docker logs mariadb 2>&1 | grep -E "01-revoke|ERROR"                          # Git Bash
docker logs mariadb 2>&1 | Select-String "01-revoke|ERROR"                     # PowerShell

# 3. MariaDB 준비 확인
# - STATUS가 (healthy)가 될 때까지 기다린 뒤 다음 단계 진행 (처음에는 30초 정도)
# - 로그에 "ready for connections"가 보이면 준비 완료
docker ps
docker logs mariadb

# 4. 앱 이미지 빌드
# - 처음에는 Maven 라이브러리 다운로드로 몇 분 걸림
docker build -t boot-mariadb-app .

# (재실행 시 기존 앱 컨테이너 정리)
docker rm -f boot-app

# 5. SpringBoot 실행
# - -d	백그라운드 실행
# - --name boot-app	컨테이너 이름
# - --network boot-net: MariaDB와 같은 네트워크
# - -p 8092:8080: 08단계(8091)와 겹치지 않도록 8092 사용
# - --env-file .env: DB 이름, 계정, 비밀번호 전달 (MariaDB와 같은 파일)
# - DB_HOST=mariadb는 Dockerfile의 ENV 기본값으로 이미 들어 있음
# - --restart unless-stopped: DB가 늦게 준비돼서 앱이 종료되더라도 자동으로 다시 시작
# (Git Bash)
docker run -d \
  --name boot-app \
  --network boot-net \
  -p 8092:8080 \
  --restart unless-stopped \
  --env-file .env \
  boot-mariadb-app

# 6. 앱 상태 확인
# - 로그에 "Started DemoApplication"과 "create table memo" SQL이 보이면 정상
docker ps
docker logs boot-app

# 7. 테스트 (브라우저)
http://localhost:8092/
http://localhost:8092/memos
http://localhost:8092/actuator/health      # components.db.status가 UP이면 DB 연결 정상

# 8. 테스트 (PowerShell: 메모 저장, 조회, 삭제)
Invoke-RestMethod -Method Post -Uri http://localhost:8092/memos -ContentType "application/json" -Body '{"content":"first memo"}'
Invoke-RestMethod http://localhost:8092/memos
Invoke-RestMethod -Method Delete -Uri http://localhost:8092/memos/1

# (Git Bash에서는 curl 사용)
curl -X POST http://localhost:8092/memos -H "Content-Type: application/json" -d '{"content":"first memo"}'
curl http://localhost:8092/memos

# 9. DB에 실제로 저장됐는지 확인 (boot 계정으로 접속)
# - 비밀번호를 물어보면 .env의 MARIADB_PASSWORD 입력
docker exec -it mariadb mariadb -uboot -p boot_db -e "SELECT * FROM memo;"

# 10. DROP 권한이 회수됐는지 확인
# - GRANT 목록에 DROP이 없으면 정상
# - 두 번째 명령은 "DROP command denied" 오류가 나야 정상
docker exec -it mariadb mariadb -uboot -p -e "SHOW GRANTS;"
docker exec -it mariadb mariadb -uboot -p boot_db -e "DROP TABLE memo;"

# 11. 데이터 유지 확인
# - 두 컨테이너를 지우고 다시 실행해도 메모가 남아 있으면 볼륨이 정상 동작하는 것
docker rm -f boot-app mariadb
#   → 2단계, 5단계 명령을 다시 실행한 뒤 http://localhost:8092/memos 확인


# ==========================================================
# 처음부터 다시 하기 (데이터 전부 삭제)
# - .env 값이나 01-revoke-drop.sql을 바꿨다면 볼륨을 지워야 다시 적용됨
# ==========================================================
docker rm -f boot-app mariadb
docker volume rm boot-mariadb-data
#   → 2단계부터 다시 실행
# (Git Bash) MariaDB 실행
MSYS_NO_PATHCONV=1 docker run -d \
  --name mariadb \
  --network boot-net \
  -p 127.0.0.1:3306:3306 \
  --restart unless-stopped \
  --env-file .env \
  -v boot-mariadb-data:/var/lib/mysql \
  -v "$(pwd -W)/db/init:/docker-entrypoint-initdb.d" \
  --health-cmd "healthcheck.sh --connect --innodb_initialized" \
  --health-interval 10s \
  --health-start-period 30s \
  mariadb:10.11.18 \
  --character-set-server=utf8mb4 \
  --collation-server=utf8mb4_unicode_ci

docker exec -it mariadb mariadb -uboot -p boot_db -e "SELECT * FROM memo;"

# (Git Bash) SpringBoot 실행
docker run -d --name boot-app --network boot-net -p 8092:8080 --restart unless-stopped --env-file .env boot-mariadb-app

curl http://localhost:8092/memos

# ==========================================================
# 전부 정리
# ==========================================================
docker rm -f boot-app mariadb
docker volume rm boot-mariadb-data
docker network rm boot-net
docker rmi boot-mariadb-app
