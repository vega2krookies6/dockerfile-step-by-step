
# (재실행 시 기존 컨테이너 정리)
# - 같은 이름(node-app)의 컨테이너가 있으면 run 단계에서 이름 충돌 오류 발생
docker rm -f node-app

# 이미지 빌드
# - -t: 이미지 이름(태그) 지정
# - 현재 폴더에 Dockerfile, package.json, app.js, healthcheck.js가 있어야 함
docker build -t my-node-app .

# 컨테이너 실행
# - -d: 백그라운드 실행, -p 3000:3000: 내 PC 3000 포트를 컨테이너 3000 포트에 연결
# - --name: 아래 inspect, exec 명령에서 이 이름(node-app)으로 컨테이너 지정
docker run -d -p 3000:3000 --name node-app my-node-app

# http://localhost:3000/health 에서 확인
# - app.js에 /health 경로가 구현되어 있어야 응답이 나옴
# - 실행 직후 상태는 starting이며, 첫 검사(약 30초 후)가 끝나야 healthy로 바뀜

=====================================================
# powershell
# 건강 상태 정보 확인
# - 현재 상태(Status), 연속 실패 횟수(FailingStreak), 최근 검사 기록(Log)을 JSON으로 출력
docker inspect --format='{{json .State.Health}}' node-app

# 건강 상태 로그 확인
# - 최근 검사 결과를 "종료 코드 : 출력 내용" 형태로 표시 (0이면 성공, 1이면 실패)
# - 실패 원인을 찾을 때 Output 내용을 보면 도움이 됨
docker inspect --format='{{range .State.Health.Log}}{{.ExitCode}} : {{.Output}}{{end}}' node-app

# 더 간단한 버전
# - 상태값만 출력: starting(검사 대기 중), healthy(정상), unhealthy(비정상)
docker inspect --format="Health Status: {{.State.Health.Status}}" node-app

# bash

# 3. 건강 체크 스크립트 수동 실행
# - HEALTHCHECK가 하는 일을 직접 실행해 결과를 바로 확인
# - 컨테이너 이름은 위에서 지정한 node-app 사용
docker exec node-app node healthcheck.js
echo $?  # 종료 코드 확인 (0이면 성공)

# 4. 건강 상태 확인
# - 상태값만 출력: starting, healthy, unhealthy 중 하나
docker inspect --format='{{.State.Health.Status}}' node-app

# 5. 상세 건강 정보
# - inspect 전체 출력에서 "Health" 부분과 그 아래 15줄만 추려서 표시
# - grep은 bash 명령이라 PowerShell에서는 Select-String "Health" -Context 0,15 사용
docker inspect node-app | grep -A 15 "Health"
