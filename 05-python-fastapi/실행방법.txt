# (재실행 시 기존 컨테이너 정리)
# - 같은 이름(my-fastapi)의 컨테이너가 있으면 run 단계에서 이름 충돌 오류 발생
# docker rm -f my-fastapi

# 2. Docker 이미지 빌드
# - -t: 이미지 이름(태그) 지정, 버전을 붙이려면 fastapi-app:1.0 형식
# - 마지막의 .은 현재 폴더를 빌드 컨텍스트로 사용
# - 현재 폴더에 Dockerfile, requirements.txt, app/main.py가 있어야 함
docker build -t fastapi-app .

# 3. 컨테이너 실행
# - -d: 백그라운드 실행 (터미널을 점유하지 않음)
# - -p 8000:8000: 내 PC의 8000 포트를 컨테이너의 8000 포트(uvicorn)에 연결
# - --name: 컨테이너 이름 지정, 이후 stop/rm/logs 등에서 이 이름으로 관리
docker run -d -p 8000:8000 --name my-fastapi fastapi-app

# 테스트 방법 (브라우저 주소창에 입력)
# - 각 경로는 app/main.py에 구현되어 있어야 응답이 나옴
# - 접속이 안 되면 docker ps로 실행 상태, docker logs my-fastapi로 로그 확인
http://localhost:8000/              # 기본 경로, 앱이 정상 실행되는지 확인
http://localhost:8000/hello/World   # 경로 변수 테스트, World 자리에 다른 이름을 넣어도 됨
http://localhost:8000/health        # 상태 확인용 경로 (헬스체크에 활용)
http://localhost:8000/info          # 앱 정보 확인용 경로
http://localhost:8000/docs          # FastAPI가 자동 생성하는 API 문서 (Swagger UI)
