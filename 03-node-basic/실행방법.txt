# (재실행 시 기존 컨테이너 정리)
# - 같은 이름(node-app)의 컨테이너가 있으면 run 단계에서 이름 충돌 오류 발생
# - 처음 실행할 때는 "No such container" 메시지가 나와도 무시해도 됨
# docker rm -f node-app

# 이미지 빌드
# - -t: 이미지 이름(태그) 지정, 버전을 붙이려면 my-node-app:1.0 형식
# - 마지막의 .은 현재 폴더를 빌드 컨텍스트로 사용
# - 현재 폴더에 Dockerfile, package.json, app.js가 있어야 함
docker build -t my-node-app .

# 컨테이너 실행
# - -d: 백그라운드 실행 (터미널을 점유하지 않음)
# - -p 3000:3000: 내 PC의 3000 포트를 컨테이너의 3000 포트(Node 앱)에 연결
# - --name: 컨테이너 이름 지정, 이후 stop/rm/logs 등에서 이 이름으로 관리
docker run -d -p 3000:3000 --name node-app my-node-app

# http://localhost:3000 에서 확인
# - 접속이 안 되면 docker ps로 실행 상태, docker logs node-app으로 로그 확인
# - 앱이 바로 종료됐다면 app.js 오류이거나 PORT 환경 변수를 읽지 않는 경우가 많음
