# 이미지 빌드
# - -t: 이미지 이름(태그) 지정, 버전을 붙이려면 my-simple-website:1.0 형식
# - 마지막의 .은 현재 폴더를 빌드 컨텍스트로 사용 (Dockerfile과 index.html이 있어야 함)
docker build -t my-simple-website .

# 컨테이너 실행
# - -d: 백그라운드 실행 (터미널을 점유하지 않음)
# - -p 8087:80: 내 PC의 8087 포트를 컨테이너의 80 포트(Nginx)에 연결
# - --name: 컨테이너 이름 지정, 이후 stop/rm 등에서 이 이름으로 관리
docker run -d -p 8087:80 --name my-web my-simple-website

# 웹사이트 확인: http://localhost:8087
# - 접속이 안 되면 docker ps로 실행 상태, 
# 로그 확인
docker logs my-web
